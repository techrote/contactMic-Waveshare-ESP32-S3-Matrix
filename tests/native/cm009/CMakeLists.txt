cmake_minimum_required(VERSION 3.16)
project(contactmic_usb_native LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(USB_CORE ${CMAKE_CURRENT_LIST_DIR}/../../../firmware-usb/components/contactmic_usb)
add_executable(cm009_core test_usb.c ${USB_CORE}/cm_usb.c ${USB_CORE}/cm_usb_descriptors.c)
target_include_directories(cm009_core PRIVATE ${USB_CORE}/include)
target_compile_options(cm009_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(cm009_core PRIVATE m)
add_test(NAME native.cm009_core COMMAND cm009_core)
add_test(NAME native.cm009_long_clocks COMMAND cm009_core long)
set_tests_properties(native.cm009_long_clocks PROPERTIES TIMEOUT 120)
add_executable(cm009_adapter test_adapter.c ${USB_CORE}/cm_usb.c ${USB_CORE}/cm_usb_descriptors.c)
target_include_directories(cm009_adapter PRIVATE stubs ${USB_CORE}/include)
target_compile_definitions(cm009_adapter PRIVATE CMU_SOURCE_REVISION="aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa")
target_compile_options(cm009_adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(cm009_adapter PRIVATE m)
add_test(NAME native.cm009_adapter COMMAND cm009_adapter)
set(ACQUISITION ${CMAKE_CURRENT_LIST_DIR}/../../../firmware/components/contactmic_acquisition)
add_executable(cm009_acquisition_pipeline test_acquisition_pipeline.c
    ${USB_CORE}/cm_usb.c ${ACQUISITION}/contactmic_acquisition.c)
target_include_directories(cm009_acquisition_pipeline PRIVATE ${USB_CORE}/include ${ACQUISITION}/include)
target_compile_options(cm009_acquisition_pipeline PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(cm009_acquisition_pipeline PRIVATE m)
add_test(NAME native.cm009_acquisition_pipeline COMMAND cm009_acquisition_pipeline)

# Verify the accounting boundary in the production core, not a counter fake.
add_executable(cm009_source_ledger test_source_ledger.c ${ACQUISITION}/contactmic_acquisition.c)
target_include_directories(cm009_source_ledger PRIVATE ${ACQUISITION}/include)
target_compile_options(cm009_source_ledger PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME native.cm009_source_ledger COMMAND cm009_source_ledger)

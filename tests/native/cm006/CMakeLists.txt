# Can also be configured alone for bounded, source-only CM-006 verification.
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(contactmic_cm006 LANGUAGES C)
    enable_testing()
endif()
set(CM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(CM_COMPONENTS "${CM_ROOT}/firmware/components")
set(CM_CORE "${CM_COMPONENTS}/contactmic_acquisition/contactmic_acquisition.c")
set(CM_SESSION_SOURCES ${CM_CORE}
    "${CM_COMPONENTS}/contactmic_session/contactmic_session.c"
    "${CM_COMPONENTS}/contactmic_ble/contactmic_transport.c"
    "${CM_COMPONENTS}/contactmic_protocol/contactmic_protocol.c")

function(cm006_target target)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    foreach(component IN ITEMS acquisition board protocol ble session)
        target_include_directories(${target} PRIVATE "${CM_COMPONENTS}/contactmic_${component}/include")
    endforeach()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    # Keep executable placement identical for standalone and parent builds.
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    add_test(NAME native.${target} COMMAND ${target})
    set_tests_properties(native.${target} PROPERTIES TIMEOUT 30)
endfunction()

add_executable(cm006_session ../session.c ${CM_SESSION_SOURCES})
cm006_target(cm006_session)
add_executable(cm006_acquisition_idf ../acquisition_idf.c ${CM_CORE})
target_include_directories(cm006_acquisition_idf PRIVATE ../idf_shim)
cm006_target(cm006_acquisition_idf)
add_executable(cm006_regressions ../acquisition_regressions.c ${CM_CORE})
cm006_target(cm006_regressions)
add_executable(cm006_simulator ../cm006_simulator.c ${CM_SESSION_SOURCES})
cm006_target(cm006_simulator)
if(UNIX)
    find_package(Threads REQUIRED)
    add_executable(cm006_threads ../acquisition_threads.c ${CM_CORE})
    target_link_libraries(cm006_threads PRIVATE Threads::Threads)
    cm006_target(cm006_threads)
endif()
